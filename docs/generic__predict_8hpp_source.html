54template <typename MotionModel_, typename CovarianceMatrixPolicy_>
55inline void Predict<MotionModel_, CovarianceMatrixPolicy_>::run(const value_type dt,
56 const InformationFilterType& filter,
57 const EgoMotionType& egoMotion)
58{
59 // Access to MotionModel internals
60 auto& underlying = static_cast<MotionModel_&>(*this);
61 auto& Y = underlying.getCovForInternalUse();
62
63 // prepare covariance prediction based on egomotion compensated state
64 static typename BasePredictCommon::Storage data{};
65
66 // Step 1: Transform from information space to state space for state prediction
67 underlying.convertStateVecIntoStateSpace();
68
69 // Step 2: Run state prediction in state space (existing code)
70 BasePredictCommon::run(data, dt, egoMotion);
71
72 // Step 3: Covariance prediction (Y_k → Y_k+1)
73 // NOTE: This MUST happen BEFORE convertStateVecIntoInformationSpace() because
74 // the transformation needs the NEW information matrix Y_k+1
75 if constexpr (CovarianceMatrixPolicy_::is_factored)
76 {
77 if (egoMotion.getDisplacementCog().vec.isZeros())
78 {
79 filter.predictCovariance(Y, data.AGo, data.G, data.Q);
80 }
81 else
82 {
83 filter.predictCovariance(Y, data.AGo, data.Gstar, data.Qstar);
84 }
85 }
86 else
87 {
88 if (egoMotion.getDisplacementCog().vec.isZeros()) // TODO(matthias): is this really correct, Yes: because we are checking if
89 // the displacement vector is all zeros without tolerances!
90 {
91 // filter prediction step with zero ego motion
92 filter.predictCovariance(Y, data.A, data.G, data.Q);
93 }
94 else
95 {
96 using StateCov = typename MotionModel_::StateCov;
97 using StateMatrix = typename MotionModel_::StateMatrix;
98 // Y = inv(Go*P*Go.T + Ge*Pe*Ge.T) with Po=inv(Y) can be solved directly in information space by
99 // applying two steps
100 // 1. Handle the State Transition
101 // Ys = inv(Go*P*Go.T) = inv(Go).T * Y * inv(Go)
102 // 2. Handle the Additive Noise Q=Ge*Pe*Ge.T
103 // Y = inv(Ps + Q), solved via H*Y = Ys with H = I + Ys*Q using QR (mirrors
104 // InformationFilter::predictCovariance's own non-factored time-update branch for G*Q*G',
105 // see information_filter.hpp) instead of inverting Pe via the Woodbury identity: Pe is
106 // structurally rank-deficient whenever the ego-motion output dimension exceeds its
107 // independent noise sources, so inverting it directly is not robust.
108
109 // step 1: Handle the State Transition (using linear solver)
110 StateMatrix Z_T = data.Go.transpose().qrSolve(Y); // solve Go.T * Z_transpose = Y
111 auto Ytr = StateCov{std::move(data.Go.transpose().qrSolve(Z_T.transpose()))}; // solve Go.T * Y_tr = Z
112 Ytr.symmetrize();
113
114 // step 2: Handle the Additive Noise Ge*Pe*Ge.T on the transformed Ytr (never inverts Pe)
115 auto Mtr = Ytr; // state-transitioned copy consumed by the QR solve below
116 const auto H = math::SquareMatrix<value_type, MotionModel_::NUM_STATE_VARIABLES>(
117 StateMatrix::Identity() + Mtr * (data.Ge * egoMotion.getDisplacementCog().cov * data.Ge.transpose()));
118 math::SquareMatrix compensated = H.qrSolve(std::move(Mtr));
119 compensated.symmetrize();
120
121 // prevent destroying the Information matrix, e.g. removing information from a zero Y matrix (no information)
122 if (compensated.isPositiveSemiDefinite())
123 {
124 Ytr = StateCov{std::move(compensated)};
125
126 // final filter prediction step with compensated Ytr
127 filter.predictCovariance(Ytr, data.A, data.G, data.Q);
128
129 // prevent destroying the Information matrix, e.g. removing information from a zero Y matrix (no information)
130 if (Ytr.isPositiveSemiDefinite())
131 {
132 Y = std::move(Ytr);
133 }
134 }
135 }
136 }
137
138 // Step 4: Transform state to information vector using the NEW information matrix Y_k+1
139 underlying.convertStateVecIntoInformationSpace();
140}
