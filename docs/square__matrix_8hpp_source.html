68template <typename ValueType_, sint32 Size_, bool IsRowMajor_>
69inline auto SquareMatrix<ValueType_, Size_, IsRowMajor_>::determinant() const -> ValueType_
70{
71 // Create a copy of the matrix for LU decomposition
72 SquareMatrix<ValueType_, Size_, IsRowMajor_> luMatrix{*this};
73 sint32 permutationCount{0};
74
75 // Perform LU decomposition with partial pivoting
76 for (auto k = 0; k < Size_; ++k)
77 {
78 // Partial pivoting: find the row with maximum element in current column
79 auto maxRow = k;
80 ValueType_ maxVal = std::abs(luMatrix.at_unsafe(k, k));
81
82 for (auto i = k + 1; i < Size_; ++i)
83 {
84 const auto absVal = std::abs(luMatrix.at_unsafe(i, k));
85 if (absVal > maxVal)
86 {
87 maxVal = absVal;
88 maxRow = i;
89 }
90 }
91
92 // Swap rows if necessary
93 if (maxRow != k)
94 {
95 for (auto j = 0; j < Size_; ++j)
96 {
97 std::swap(luMatrix.at_unsafe(k, j), luMatrix.at_unsafe(maxRow, j));
98 }
99 permutationCount++;
100 }
101
102 // Check for singular matrix
103 if (std::abs(luMatrix.at_unsafe(k, k)) < std::numeric_limits<ValueType_>::epsilon())
104 {
105 return static_cast<ValueType_>(0);
106 }
107
108 // Perform elimination for rows below the current one
109 for (auto i = k + 1; i < Size_; ++i)
110 {
111 const auto factor = luMatrix.at_unsafe(i, k) / luMatrix.at_unsafe(k, k);
112
113 for (auto j = k; j < Size_; ++j)
114 {
115 luMatrix.at_unsafe(i, j) -= factor * luMatrix.at_unsafe(k, j);
116 }
117 }
118 }
119
120 // Calculate determinant as product of diagonal elements, multiplied by (-1)^permutationCount
121 ValueType_ det{1};
122 for (auto i = 0; i < Size_; ++i)
123 {
124 det *= luMatrix.at_unsafe(i, i);
125 }
126
127 // Apply sign based on number of row permutations
128 if (permutationCount % 2 != 0)
129 {
130 det = -det;
131 }
132
133 return det;
134}

156{

160 for (sint32 col = row + 1; col < Size_; ++col)

198template <typename ValueType_, sint32 Size_, bool IsRowMajor_>
199inline auto SquareMatrix<ValueType_, Size_, IsRowMajor_>::isPositiveSemiDefinite() const -> bool
200{
201 // A strict Cholesky (decomposeLLT/isPositiveDefinite) rejects singular PSD matrices (zero
202 // eigenvalue -> zero pivot), which is wrong for a semi-definiteness test and would freeze the
203 // covariance/information prediction in the filter guards. Instead run a tolerance-aware LDL^T:
204 // a symmetric matrix is positive semi-definite if it admits an LDL^T with every pivot >= 0.
205 if (!isSymmetric())
206 {
207 return false;
208 }
209
210 // scale-relative tolerance from the largest diagonal magnitude (all diagonals of a PSD matrix
211 // are non-negative, so this bounds the spectral scale)
212 ValueType_ scale = static_cast<ValueType_>(0);
213 for (auto idx = 0; idx < Size_; ++idx)
214 {
215 scale = std::max(scale, std::abs(this->at_unsafe(idx, idx)));
216 }
217 const ValueType_ tolerance =
218 std::numeric_limits<ValueType_>::epsilon() * static_cast<ValueType_>(Size_) * std::max(scale, static_cast<ValueType_>(1));
219
220 DiagonalMatrix<ValueType_, Size_> D{};
221 TriangularMatrix<ValueType_, Size_, true, IsRowMajor_> L{};
222 for (auto j = 0; j < Size_; ++j)
223 {
224 ValueType_ pivot = this->at_unsafe(j, j);
225 for (auto k = 0; k < j; ++k)
226 {
227 pivot -= D.at_unsafe(k) * L.at_unsafe(j, k) * L.at_unsafe(j, k);
228 }
229 if (pivot < -tolerance)
230 {
231 return false; // strictly negative pivot -> indefinite / not positive semi-definite
232 }
233 D.at_unsafe(j) = pivot;
234 L.at_unsafe(j, j) = static_cast<ValueType_>(1);
235
236 for (auto i = j + 1; i < Size_; ++i)
237 {
238 ValueType_ off = this->at_unsafe(i, j);
239 for (auto k = 0; k < j; ++k)
240 {
241 off -= D.at_unsafe(k) * L.at_unsafe(i, k) * L.at_unsafe(j, k);
242 }
243 if (pivot > tolerance)
244 {
245 L.at_unsafe(i, j) = off / pivot;
246 }
247 else
248 {
249 // Zero pivot: for a genuinely PSD matrix the corresponding Schur-complement column must
250 // vanish. A non-negligible off-diagonal here means the matrix is not PSD.
251 if (std::abs(off) > tolerance)
252 {
253 return false;
254 }
255 L.at_unsafe(i, j) = static_cast<ValueType_>(0);
256 }
257 }
258 }
259 return true;
260}
