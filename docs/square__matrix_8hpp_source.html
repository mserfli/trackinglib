62template <typename ValueType_, sint32 Size_, bool IsRowMajor_>
63inline auto SquareMatrix<ValueType_, Size_, IsRowMajor_>::determinant() const -> ValueType_
64{
65 // Create a copy of the matrix for LU decomposition
66 SquareMatrix<ValueType_, Size_, IsRowMajor_> luMatrix{*this};
67 sint32 permutationCount{0};
68
69 // Perform LU decomposition with partial pivoting
70 for (auto k = 0; k < Size_; ++k)
71 {
72 // Partial pivoting: find the row with maximum element in current column
73 auto maxRow = k;
74 ValueType_ maxVal = std::abs(luMatrix.at_unsafe(k, k));
75
76 for (auto i = k + 1; i < Size_; ++i)
77 {
78 const auto absVal = std::abs(luMatrix.at_unsafe(i, k));
79 if (absVal > maxVal)
80 {
81 maxVal = absVal;
82 maxRow = i;
83 }
84 }
85
86 // Swap rows if necessary
87 if (maxRow != k)
88 {
89 for (auto j = 0; j < Size_; ++j)
90 {
91 std::swap(luMatrix.at_unsafe(k, j), luMatrix.at_unsafe(maxRow, j));
92 }
93 permutationCount++;
94 }
95
96 // Check for singular matrix
97 if (std::abs(luMatrix.at_unsafe(k, k)) < std::numeric_limits<ValueType_>::epsilon())
98 {
99 return static_cast<ValueType_>(0);
100 }
101
102 // Perform elimination for rows below the current one
103 for (auto i = k + 1; i < Size_; ++i)
104 {
105 const auto factor = luMatrix.at_unsafe(i, k) / luMatrix.at_unsafe(k, k);
106
107 for (auto j = k; j < Size_; ++j)
108 {
109 luMatrix.at_unsafe(i, j) -= factor * luMatrix.at_unsafe(k, j);
110 }
111 }
112 }
113
114 // Calculate determinant as product of diagonal elements, multiplied by (-1)^permutationCount
115 ValueType_ det{1};
116 for (auto i = 0; i < Size_; ++i)
117 {
118 det *= luMatrix.at_unsafe(i, i);
119 }
120
121 // Apply sign based on number of row permutations
122 if (permutationCount % 2 != 0)
123 {
124 det = -det;
125 }
126
127 return det;
128}

192template <typename ValueType_, sint32 Size_, bool IsRowMajor_>
193inline auto SquareMatrix<ValueType_, Size_, IsRowMajor_>::isPositiveSemiDefinite() const -> bool
194{
195 // A strict Cholesky (decomposeLLT/isPositiveDefinite) rejects singular PSD matrices (zero
196 // eigenvalue -> zero pivot), which is wrong for a semi-definiteness test and would freeze the
197 // covariance/information prediction in the filter guards. Instead run a tolerance-aware LDL^T:
198 // a symmetric matrix is positive semi-definite if it admits an LDL^T with every pivot >= 0.
199 if (!isSymmetric())
200 {
201 return false;
202 }
203
204 // scale-relative tolerance from the largest diagonal magnitude (all diagonals of a PSD matrix
205 // are non-negative, so this bounds the spectral scale)
206 ValueType_ scale = static_cast<ValueType_>(0);
207 for (auto idx = 0; idx < Size_; ++idx)
208 {
209 scale = std::max(scale, std::abs(this->at_unsafe(idx, idx)));
210 }
211 const ValueType_ tolerance =
212 std::numeric_limits<ValueType_>::epsilon() * static_cast<ValueType_>(Size_) * std::max(scale, static_cast<ValueType_>(1));
213
214 DiagonalMatrix<ValueType_, Size_> D{};
215 TriangularMatrix<ValueType_, Size_, true, IsRowMajor_> L{};
216 for (auto j = 0; j < Size_; ++j)
217 {
218 ValueType_ pivot = this->at_unsafe(j, j);
219 for (auto k = 0; k < j; ++k)
220 {
221 pivot -= D.at_unsafe(k) * L.at_unsafe(j, k) * L.at_unsafe(j, k);
222 }
223 if (pivot < -tolerance)
224 {
225 return false; // strictly negative pivot -> indefinite / not positive semi-definite
226 }
227 D.at_unsafe(j) = pivot;
228 L.at_unsafe(j, j) = static_cast<ValueType_>(1);
229
230 for (auto i = j + 1; i < Size_; ++i)
231 {
232 ValueType_ off = this->at_unsafe(i, j);
233 for (auto k = 0; k < j; ++k)
234 {
235 off -= D.at_unsafe(k) * L.at_unsafe(i, k) * L.at_unsafe(j, k);
236 }
237 if (pivot > tolerance)
238 {
239 L.at_unsafe(i, j) = off / pivot;
240 }
241 else
242 {
243 // Zero pivot: for a genuinely PSD matrix the corresponding Schur-complement column must
244 // vanish. A non-negligible off-diagonal here means the matrix is not PSD.
245 if (std::abs(off) > tolerance)
246 {
247 return false;
248 }
249 L.at_unsafe(i, j) = static_cast<ValueType_>(0);
250 }
251 }
252 }
253 return true;
254}
