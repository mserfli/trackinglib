248 [[nodiscard]] auto toSensorFrame(const StateVec& state) const -> StateVec
249 {
250 StateVec transformed = state;
251 if constexpr (motion::has_position_v<StateDef>)
252 {
253 const auto p = getSensorPose().positionToSensorFrame(state.at_unsafe(StateDef::X), state.at_unsafe(StateDef::Y));
254 transformed.at_unsafe(StateDef::X) = p.x();
255 transformed.at_unsafe(StateDef::Y) = p.y();
256 }
257 if constexpr (motion::has_velocity_v<StateDef>)
258 {
259 const auto v = getSensorPose().directionToSensorFrame(state.at_unsafe(StateDef::VX), state.at_unsafe(StateDef::VY));
260 transformed.at_unsafe(StateDef::VX) = v.x();
261 transformed.at_unsafe(StateDef::VY) = v.y();
262 }
263 if constexpr (motion::has_acceleration_v<StateDef>)
264 {
265 const auto a = getSensorPose().directionToSensorFrame(state.at_unsafe(StateDef::AX), state.at_unsafe(StateDef::AY));
266 transformed.at_unsafe(StateDef::AX) = a.x();
267 transformed.at_unsafe(StateDef::AY) = a.y();
268 }
269 return transformed;
270 }

281 void applyJacobianFrameRotation(JacobianMatrix& jacobian) const
282 {
283 if constexpr (motion::has_position_v<StateDef>)
284 {
285 getSensorPose().rotateJacobianColumns(jacobian, StateDef::X, StateDef::Y);
286 }
287 if constexpr (motion::has_velocity_v<StateDef>)
288 {
289 getSensorPose().rotateJacobianColumns(jacobian, StateDef::VX, StateDef::VY);
290 }
291 if constexpr (motion::has_acceleration_v<StateDef>)
292 {
293 getSensorPose().rotateJacobianColumns(jacobian, StateDef::AX, StateDef::AY);
294 }
295 }
