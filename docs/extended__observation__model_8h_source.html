136 static auto FromLists(const std::initializer_list<value_type>& vecList,

313 [[nodiscard]] auto toSensorFrame(const StateVec& state) const -> StateVec
314 {
315 StateVec transformed = state;
316 if constexpr (motion::has_position_v<StateDef>)
317 {
318 const auto p = getSensorPose().positionToSensorFrame(state.at_unsafe(StateDef::X), state.at_unsafe(StateDef::Y));
319 transformed.at_unsafe(StateDef::X) = p.x();
320 transformed.at_unsafe(StateDef::Y) = p.y();
321 }
322 if constexpr (motion::has_velocity_v<StateDef>)
323 {
324 const auto v = getSensorPose().directionToSensorFrame(state.at_unsafe(StateDef::VX), state.at_unsafe(StateDef::VY));
325 transformed.at_unsafe(StateDef::VX) = v.x();
326 transformed.at_unsafe(StateDef::VY) = v.y();
327 }
328 if constexpr (motion::has_acceleration_v<StateDef>)
329 {
330 const auto a = getSensorPose().directionToSensorFrame(state.at_unsafe(StateDef::AX), state.at_unsafe(StateDef::AY));
331 transformed.at_unsafe(StateDef::AX) = a.x();
332 transformed.at_unsafe(StateDef::AY) = a.y();
333 }
334 return transformed;
335 }

346 void applyJacobianFrameRotation(JacobianMatrix& jacobian) const
347 {
348 if constexpr (motion::has_position_v<StateDef>)
349 {
350 getSensorPose().rotateJacobianColumns(jacobian, StateDef::X, StateDef::Y);
351 }
352 if constexpr (motion::has_velocity_v<StateDef>)
353 {
354 getSensorPose().rotateJacobianColumns(jacobian, StateDef::VX, StateDef::VY);
355 }
356 if constexpr (motion::has_acceleration_v<StateDef>)
357 {
358 getSensorPose().rotateJacobianColumns(jacobian, StateDef::AX, StateDef::AY);
359 }
360 }
