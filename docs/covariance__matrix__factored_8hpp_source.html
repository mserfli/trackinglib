97template <typename ValueType_, sint32 Size_>
98inline auto CovarianceMatrixFactored<ValueType_, Size_>::at_unsafe(sint32 row, sint32 col) const -> ValueType_
99{
100 // cov(row,col) == cov(col,row), so we swap row and col if row > col
101 if (row > col)
102 {
103 std::swap(row, col);
104 }
105
106 ValueType_ result{};
107 // calc row, col element of _u * _d * _u.transpose()
108 // calc relevant elements of u*d to be rhs multiplied with uT(col::, col)==u(col, col::)
109 Vector<ValueType_, Size_> ud{};
110 for (auto i = col; i < Size_; ++i)
111 {
112 ud.at_unsafe(i) = _u.at_unsafe(row, i) * _d.at_unsafe(i);
113 }
114 MatrixColumnView<ValueType_, Size_, 1, true> const udView{ud, 0, col, Size_ - 1};
115 MatrixRowView<ValueType_, Size_, Size_, true> const uTView{_u, col, col, Size_ - 1};
116 result = uTView * udView; // calc the scalar product of ud*uT on relevant elements
117 return result;
118}

139template <typename ValueType_, sint32 Size_>
140inline auto CovarianceMatrixFactored<ValueType_, Size_>::inverse() const -> tl::expected<CovarianceMatrixFactored, Errors>
141{
142 // we use the transpose of the input matrix to get a column major matrix
143 const auto composed = tracking::math::SquareMatrix<ValueType_, Size_, false>{this->operator()().transpose()};
144 // we decompose the input matrix into LDLt form, with L being a column major lower triangular matrix
145 const auto ldlt = composed.decomposeLDLT();
146 if (ldlt.has_value())
147 {
148 const auto [l, d] = ldlt.value();
149 // we calc the inverse of L and D and map the inv(L).transpose() to U being again a row major upper triangular matrix
150 // the resulting UDUt matrix describes the covariance matrix in information form, i.e. the inverse covariance matrix
151 return CovarianceMatrixFactored{std::move(l.inverse().transpose()), std::move(d.inverse())};
152 }
153
154 return tl::unexpected<Errors>{ldlt.error()};
155}
