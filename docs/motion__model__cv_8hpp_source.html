34 const value_type deltaXEgo = egoMotion.getDisplacementCog().vec.at_unsafe(EgoMotionType::DS_X);
35 const value_type deltaYEgo = egoMotion.getDisplacementCog().vec.at_unsafe(EgoMotionType::DS_Y);
36 const value_type distCog2Ego = egoMotion.getGeometry().distCog2Ego;

38 Go.setZeros();

40 Go.at_unsafe(X, Y) = sinDeltaPsiEgo;

42 Go.at_unsafe(Y, Y) = cosDeltaPsiEgo;
43 Go.at_unsafe(VX, VX) = cosDeltaPsiEgo;
44 Go.at_unsafe(VX, VY) = sinDeltaPsiEgo;
45 Go.at_unsafe(VY, VX) = -sinDeltaPsiEgo;
46 Go.at_unsafe(VY, VY) = cosDeltaPsiEgo;
47

52 Ge.at_unsafe(X, EgoMotionType::DS_Y) = -sinDeltaPsiEgo;

55 Ge.at_unsafe(Y, EgoMotionType::DS_Y) = cosDeltaPsiEgo;

57 Ge.at_unsafe(VX, EgoMotionType::DS_PSI) = -(vx * sinDeltaPsiEgo) + (vy * cosDeltaPsiEgo);
58 Ge.at_unsafe(VY, EgoMotionType::DS_PSI) = -(vx * cosDeltaPsiEgo) - (vy * sinDeltaPsiEgo);
59}
