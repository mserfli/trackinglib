142 void computeJacobianSensorFrame(JacobianMatrix& jacobian,
143 const StateVec& state,
144 const typename BaseExtendedObservationModel::EgoMotionType& egoMotion) const
145 {
146 static_cast<void>(egoMotion);
147 const value_type x = state.at_unsafe(StateDef_::X);
148 const value_type y = state.at_unsafe(StateDef_::Y);
149 const value_type rangeSq = std::max((x * x) + (y * y), RANGE_SQ_MIN);
150 const value_type range = std::sqrt(rangeSq);
151
152 jacobian.setZeros();
153 jacobian.at_unsafe(MEAS_RANGE, StateDef_::X) = x / range;
154 jacobian.at_unsafe(MEAS_RANGE, StateDef_::Y) = y / range;
155 }
