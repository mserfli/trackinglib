131 auto predictMeasurementSensorFrame(
132 const StateVec& state, const typename BaseExtendedObservationModel::EgoMotionType& egoMotion) const -> MeasurementVec
133 {
134 const value_type x = state.at_unsafe(StateDef_::X);
135 const value_type y = state.at_unsafe(StateDef_::Y);
136 const value_type range = std::sqrt(std::max((x * x) + (y * y), RANGE_SQ_MIN));
137
138 MeasurementVec predicted{};
139 predicted.at_unsafe(MEAS_RANGE) = std::sqrt((x * x) + (y * y));
140 predicted.at_unsafe(MEAS_BEARING) = std::atan2(y, x);
141 if constexpr (motion::has_velocity_v<StateDef_>)
142 {
143 const value_type vx = state.at_unsafe(StateDef_::VX);
144 const value_type vy = state.at_unsafe(StateDef_::VY);
145
146 const auto egoVelSensor = this->egoVelocitySensorFrame(egoMotion);
147 const value_type vxRel = vx - egoVelSensor.x();
148 const value_type vyRel = vy - egoVelSensor.y();
149
150 predicted.at_unsafe(MEAS_DOPPLER) = ((x * vxRel) + (y * vyRel)) / range;
151 }
152 else
153 {
154 predicted.at_unsafe(MEAS_DOPPLER) = static_cast<value_type>(0.0);
155 }

169 void computeJacobianSensorFrame(JacobianMatrix& jacobian,
170 const StateVec& state,
171 const typename BaseExtendedObservationModel::EgoMotionType& egoMotion) const
172 {
173 const value_type x = state.at_unsafe(StateDef_::X);
174 const value_type y = state.at_unsafe(StateDef_::Y);
175 const value_type rangeSq = std::max((x * x) + (y * y), RANGE_SQ_MIN);
176 const value_type range = std::sqrt(rangeSq);
177
178 jacobian.setZeros();
179 jacobian.at_unsafe(MEAS_RANGE, StateDef_::X) = x / range;
180 jacobian.at_unsafe(MEAS_RANGE, StateDef_::Y) = y / range;
181 jacobian.at_unsafe(MEAS_BEARING, StateDef_::X) = -y / rangeSq;
182 jacobian.at_unsafe(MEAS_BEARING, StateDef_::Y) = x / rangeSq;
183 if constexpr (motion::has_velocity_v<StateDef_>)
184 {
185 const value_type vx = state.at_unsafe(StateDef_::VX);
186 const value_type vy = state.at_unsafe(StateDef_::VY);
187
188 const auto egoVelSensor = this->egoVelocitySensorFrame(egoMotion);
189 const value_type vxRel = vx - egoVelSensor.x();
190 const value_type vyRel = vy - egoVelSensor.y();
191 const value_type radial = (x * vxRel) + (y * vyRel);
192
193 jacobian.at_unsafe(MEAS_DOPPLER, StateDef_::X) = (vxRel / range) - ((x * radial) / (rangeSq * range));
194 jacobian.at_unsafe(MEAS_DOPPLER, StateDef_::Y) = (vyRel / range) - ((y * radial) / (rangeSq * range));
195 jacobian.at_unsafe(MEAS_DOPPLER, StateDef_::VX) = x / range;
196 jacobian.at_unsafe(MEAS_DOPPLER, StateDef_::VY) = y / range;
197 }
198 }

204 auto computeInnovation(const MeasurementVec& measurement, const MeasurementVec& predicted) const -> MeasurementVec
205 {
206 MeasurementVec innovation{measurement - predicted};
207 value_type& bearing = innovation.at_unsafe(MEAS_BEARING);
208 bearing = std::atan2(std::sin(bearing), std::cos(bearing));
209 return innovation;
