144 auto predictMeasurementSensorFrame(
145 const StateVec& state, const typename BaseExtendedObservationModel::EgoMotionType& egoMotion) const -> MeasurementVec
146 {
147 const value_type x = state.at_unsafe(StateDef_::X);
148 const value_type y = state.at_unsafe(StateDef_::Y);
149 const value_type range = std::sqrt(std::max((x * x) + (y * y), RANGE_SQ_MIN));
150
151 MeasurementVec predicted{};
152 predicted.at_unsafe(MEAS_RANGE) = std::sqrt((x * x) + (y * y));
153 predicted.at_unsafe(MEAS_BEARING) = std::atan2(y, x);
154 if constexpr (motion::has_velocity_v<StateDef_>)
155 {
156 const value_type vx = state.at_unsafe(StateDef_::VX);
157 const value_type vy = state.at_unsafe(StateDef_::VY);
158
159 const auto egoVelSensor = this->egoVelocitySensorFrame(egoMotion);
160 const value_type vxRel = vx - egoVelSensor.x();
161 const value_type vyRel = vy - egoVelSensor.y();
162
163 predicted.at_unsafe(MEAS_DOPPLER) = ((x * vxRel) + (y * vyRel)) / range;
164 }
165 else
166 {
167 predicted.at_unsafe(MEAS_DOPPLER) = static_cast<value_type>(0.0);
168 }
169 return predicted;
170 }

182 void computeJacobianSensorFrame(JacobianMatrix& jacobian,
183 const StateVec& state,
184 const typename BaseExtendedObservationModel::EgoMotionType& egoMotion) const
185 {
186 const value_type x = state.at_unsafe(StateDef_::X);
187 const value_type y = state.at_unsafe(StateDef_::Y);
188 const value_type rangeSq = std::max((x * x) + (y * y), RANGE_SQ_MIN);
189 const value_type range = std::sqrt(rangeSq);
190
191 jacobian.setZeros();
192 jacobian.at_unsafe(MEAS_RANGE, StateDef_::X) = x / range;
193 jacobian.at_unsafe(MEAS_RANGE, StateDef_::Y) = y / range;
194 jacobian.at_unsafe(MEAS_BEARING, StateDef_::X) = -y / rangeSq;
195 jacobian.at_unsafe(MEAS_BEARING, StateDef_::Y) = x / rangeSq;
196 if constexpr (motion::has_velocity_v<StateDef_>)
197 {
198 const value_type vx = state.at_unsafe(StateDef_::VX);
199 const value_type vy = state.at_unsafe(StateDef_::VY);
200
201 const auto egoVelSensor = this->egoVelocitySensorFrame(egoMotion);
202 const value_type vxRel = vx - egoVelSensor.x();
203 const value_type vyRel = vy - egoVelSensor.y();
204 const value_type radial = (x * vxRel) + (y * vyRel);
205
206 jacobian.at_unsafe(MEAS_DOPPLER, StateDef_::X) = (vxRel / range) - ((x * radial) / (rangeSq * range));
207 jacobian.at_unsafe(MEAS_DOPPLER, StateDef_::Y) = (vyRel / range) - ((y * radial) / (rangeSq * range));
208 jacobian.at_unsafe(MEAS_DOPPLER, StateDef_::VX) = x / range;
209 jacobian.at_unsafe(MEAS_DOPPLER, StateDef_::VY) = y / range;
210 }

217 auto computeInnovation(const MeasurementVec& measurement, const MeasurementVec& predicted) const -> MeasurementVec
218 {
219 MeasurementVec innovation{measurement - predicted};
220 value_type& bearing = innovation.at_unsafe(MEAS_BEARING);
221 bearing = std::atan2(std::sin(bearing), std::cos(bearing));
222 return innovation;
223 }
