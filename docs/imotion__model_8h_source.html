262 void convertStateVecIntoStateSpace()
263 {
264 const auto& y = this->getVec();
265 const auto& Y = this->getCovForInternalUse();
266 auto& x = this->getVecForInternalUse();
267 // precondition: y and Y must be in information space (Y = P^-1, y = Y * x)
268 // operation:
269 // Y * result = y
270 // result = Y^-1 * y = Y^-1 * Y * x = x
271 x = Y().qrSolve(y);
272 // postcondition: x is in state space, Y is unchanged (still in information space)
273 }

276 void convertStateVecIntoInformationSpace()
277 {
278 const auto& x = this->getVec();
279 const auto& Y = this->getCovForInternalUse();
280 auto& y = this->getVecForInternalUse();
281 // precondition: x must be in state space, Y must be in information space (Y = P^-1)
282 // operation:
283 // y = Y * x
284 y = static_cast<StateVec>(Y() * x);
285 // postcondition: y is in information space (y = Y * x), Y is unchanged (still in information space)
286 }
