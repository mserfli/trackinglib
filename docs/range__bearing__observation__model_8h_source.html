128 [[nodiscard]] auto predictMeasurementSensorFrame(
129 const StateVec& state, const typename BaseExtendedObservationModel::EgoMotionType& egoMotion) const -> MeasurementVec
130 {
131 static_cast<void>(egoMotion);
132 const value_type x = state.at_unsafe(StateDef_::X);
133 const value_type y = state.at_unsafe(StateDef_::Y);
134
135 MeasurementVec predicted{};
136 predicted.at_unsafe(MEAS_RANGE) = std::sqrt((x * x) + (y * y));
137 predicted.at_unsafe(MEAS_BEARING) = std::atan2(y, x);
138 return predicted;
139 }

151 void computeJacobianSensorFrame(JacobianMatrix& jacobian,
152 const StateVec& state,
153 const typename BaseExtendedObservationModel::EgoMotionType& egoMotion) const
154 {
155 static_cast<void>(egoMotion);
156 const value_type x = state.at_unsafe(StateDef_::X);
157 const value_type y = state.at_unsafe(StateDef_::Y);
158 const value_type rangeSq = std::max((x * x) + (y * y), RANGE_SQ_MIN);
159 const value_type range = std::sqrt(rangeSq);
160
161 jacobian.setZeros();
162 jacobian.at_unsafe(MEAS_RANGE, StateDef_::X) = x / range;
163 jacobian.at_unsafe(MEAS_RANGE, StateDef_::Y) = y / range;
164 jacobian.at_unsafe(MEAS_BEARING, StateDef_::X) = -y / rangeSq;
165 jacobian.at_unsafe(MEAS_BEARING, StateDef_::Y) = x / rangeSq;
166 }

172 [[nodiscard]] auto computeInnovation(const MeasurementVec& measurement, const MeasurementVec& predicted) const -> MeasurementVec
173 {
174 MeasurementVec innovation{measurement - predicted};
175 value_type& bearing = innovation.at_unsafe(MEAS_BEARING);
176 bearing = std::atan2(std::sin(bearing), std::cos(bearing));
177 return innovation;
178 }
